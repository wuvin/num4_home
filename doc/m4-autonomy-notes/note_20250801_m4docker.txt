# Server setup (either laptop or Jetson, if ROS 2 available)

    source /opt/ros/humble/setup.bash

    # Check zero-based server position to set in ROS_DISCOVERY_SERVER
    echo "$(hostname -I | awk '{print $1}')"
    # 192.168.0.146 (laptop default) or 192.168.1.50 (Jetson default)
    
    # Start discovery server
    fastdds discovery --server-id 0
    # -i, --server-id specifies zero-based position as unique server identifier
    # -l, --udp-address IPv4/6 or name to listen, default (0.0.0.0/::0) or all
    # -p, --udp-port namesake, default 11811
    # defaults to UDP instead of TCP (-t, --tcp-address and -q, --tcp-port)
    
# Container setup (should be ran with `--network`)

    source /opt/ros/humble/setup.bash

    # Set location of discovery server into environment
    export ROS_DISCOVERY_SERVER=<hostname>:<port>
    # e.g., 192.168.0.146:11811
    
    # Check existing topics before test topic creation
    ros2 topic list
    # only /parameter_events and /rosout by default
    
# Server new terminal

    source /opt/ros/humble/setup.bash
    
    # Set location of discovery server into environment
    export ROS_DISCOVERY_SERVER=<hostaddr>:<port>
    # e.g., 192.168.0.146:11811
    
    # Test publisher
    ros2 topic pub /test_topic std_msgs/msg/String "data: 'Hello from Laptop'" --rate 1
